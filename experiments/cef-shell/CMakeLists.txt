cmake_minimum_required(VERSION 3.21)
project(DexerionCefExperiment LANGUAGES C CXX)

if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This isolated prototype targets Linux and Windows.")
endif()
if(DEFINED CEF_ROOT)
  file(TO_CMAKE_PATH "${CEF_ROOT}" CEF_ROOT)
endif()
if(NOT DEFINED CEF_ROOT OR NOT EXISTS "${CEF_ROOT}/cmake/FindCEF.cmake")
  message(FATAL_ERROR "Set CEF_ROOT to the exact extracted CEF SDK directory.")
endif()
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox required" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
ADD_LOGICAL_TARGET(libcef_import "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
SET_CEF_TARGET_OUT_DIR()

set(host dexerion_cef_probe)
if(OS_WINDOWS)
  add_library(${host} SHARED main.cc)
  SET_LIBRARY_TARGET_PROPERTIES(${host})
  COPY_SINGLE_FILE(${host} "${CEF_BINARY_DIR}/bootstrap.exe" "${CEF_TARGET_OUT_DIR}/${host}.exe")
else()
  add_executable(${host} main.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${host})
  set_target_properties(${host} PROPERTIES
    INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE
    RUNTIME_OUTPUT_DIRECTORY "${CEF_TARGET_OUT_DIR}")
endif()
target_compile_features(${host} PRIVATE cxx_std_20)
target_link_libraries(${host} PRIVATE libcef_import libcef_dll_wrapper ${CEF_STANDARD_LIBS})
COPY_FILES("${host}" "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${CEF_TARGET_OUT_DIR}")
COPY_FILES("${host}" "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${CEF_TARGET_OUT_DIR}")
if(OS_WINDOWS)
  SET_LPAC_ACLS("${host}")
endif()
