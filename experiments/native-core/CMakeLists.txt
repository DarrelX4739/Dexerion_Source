cmake_minimum_required(VERSION 3.16)
project(dexerion_native_core LANGUAGES CXX)

include(CTest)
if(BUILD_TESTING)
  find_program(DEX_NODE_EXECUTABLE NAMES node nodejs)
  if(NOT DEX_NODE_EXECUTABLE)
    message(FATAL_ERROR "Node is required to run native/JS parity tests; install Node or set BUILD_TESTING=OFF for library-only builds.")
  endif()
endif()

function(add_policy name short_name)
  add_library(dex_${name} STATIC src/${name}.cpp)
  target_include_directories(dex_${name} PUBLIC include)
  target_compile_features(dex_${name} PUBLIC cxx_std_17)
  set_target_properties(dex_${name} PROPERTIES CXX_EXTENSIONS OFF)
  if(BUILD_TESTING)
    add_executable(dex_${short_name}_unit tests/${name}_unit.cpp)
    target_link_libraries(dex_${short_name}_unit PRIVATE dex_${name})
    add_test(NAME ${name}_unit COMMAND dex_${short_name}_unit)
    add_executable(dex_${short_name}_replay tests/${name}_replay.cpp)
    target_link_libraries(dex_${short_name}_replay PRIVATE dex_${name})
    add_test(NAME ${name}_js_parity COMMAND "${DEX_NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}_parity.cjs" $<TARGET_FILE:dex_${short_name}_replay>)
  endif()
endfunction()
